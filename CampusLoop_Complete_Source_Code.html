  32 |     n = len(poly)
  33 |     inside = False
  34 |     p1x, p1y = poly[0]
  35 |     for i in range(1, n + 1):

  38 |             if y <= max(p1y, p2y):

  52 | # -- image compression & validation helpers ---------------------------------
  53 | def process_and_compress_image(data_uri, max_dimension=1600, target_max_bytes=1024 * 1024):
  54 |     """
  55 |     Validates, corrects orientation, scales very high-res photos to max_dimension (preserving aspect ratio),
  56 |     and compresses to JPEG under target_max_bytes (~500 KB - 1 MB).
  57 |     Gracefully falls back to original data_uri if compression cannot be applied, so submissions NEVER fail.
  58 |     """
  59 |     if not data_uri or not isinstance(data_uri, str) or not data_uri.strip():
  60 |         raise ValueError("Image data is missing or invalid.")
  61 | 
  62 |     # Calculate fallback hash immediately
  63 |     fallback_hash = hashlib.sha256(data_uri.encode("utf-8")).hexdigest()
  64 | 
  65 |     try:
  66 |         if "," in data_uri:
  67 |             header, encoded = data_uri.split(",", 1)
  68 |         else:
  69 |             encoded = data_uri
  70 | 
  71 |         # Remove any whitespace or line breaks in base64 string
  72 |         encoded = re.sub(r"\s+", "", encoded)
  73 |         raw_bytes = base64.b64decode(encoded)
  74 | 
  75 |         if len(raw_bytes) < 30:
  76 |             return data_uri, fallback_hash
  77 | 
  78 |         img = Image.open(io.BytesIO(raw_bytes))
  79 | 
  80 |         # Handle EXIF orientation (transpose) so mobile camera photos remain upright
  81 |         try:
  82 |             img = ImageOps.exif_transpose(img)
  83 |         except Exception:
  84 |             pass
  85 | 
  86 |         # Convert palette/transparency to RGB for clean JPEG compression
  87 |         if img.mode in ("RGBA", "LA"):
  88 |             background = Image.new("RGB", img.size, (255, 255, 255))
  89 |             background.paste(img, mask=img.split()[-1])
  90 |             img = background
  91 |         elif img.mode != "RGB":
  92 |             img = img.convert("RGB")
  93 | 
  94 |         # Resize if exceeding max_dimension while strictly preserving aspect ratio
  95 |         width, height = img.size
  96 |         if width > max_dimension or height > max_dimension:
  97 |             scale = min(max_dimension / width, max_dimension / height)
  98 |             new_width = max(1, round(width * scale))
  99 |             new_height = max(1, round(height * scale))
 100 |             img = img.resize((new_width, new_height), Image.Resampling.LANCZOS)
 101 | 
 102 |         # Compress to JPEG with target size (~500 KB - 1 MB)
 103 |         quality = 84
 104 |         out_buf = io.BytesIO()
 105 |         img.save(out_buf, format="JPEG", quality=quality, optimize=True)
 106 | 
 107 |         # If it exceeds target_max_bytes, step down quality slightly down to 70 without making it blurry
 108 |         while len(out_buf.getvalue()) > target_max_bytes and quality > 70:
 109 |             quality -= 5
 110 |             out_buf = io.BytesIO()
 111 |             img.save(out_buf, format="JPEG", quality=quality, optimize=True)
 112 | 
 113 |         compressed_bytes = out_buf.getvalue()
 114 |         compressed_b64 = base64.b64encode(compressed_bytes).decode("ascii")
 115 |         compressed_uri = f"data:image/jpeg;base64,{compressed_b64}"
 116 |         img_hash = hashlib.sha256(compressed_bytes).hexdigest()
 117 | 
 118 |         return compressed_uri, img_hash
 119 | 
 120 |     except Exception:
 121 |         # Fail-safe: if any compression error occurs, return original data so submission is never broken
 122 |         return data_uri, fallback_hash

 674 |     async function compressImageFile(file, options = {}) {
 675 |         if (!file) {
 676 |             return { dataUrl: null, originalSize: 0, compressedSize: 0, width: 0, height: 0 };
 677 |         }
 678 |         const rawDataUrl = await readFileAsDataUrl(file);
 679 |         if (!rawDataUrl) {
 680 |             return { dataUrl: null, originalSize: file.size, compressedSize: file.size, width: 0, height: 0 };
 681 |         }
 682 | 
 683 |         try {
 684 |             const maxDimension = options.maxDimension || 1600;
 685 |             const img = new Image();
 686 |             await new Promise((resolve) => {
 687 |                 img.onload = () => resolve();
 688 |                 img.onerror = () => resolve();
 689 |                 img.src = rawDataUrl;
 690 |             });
 691 | 
 692 |             let width = img.naturalWidth || img.width;
 693 |             let height = img.naturalHeight || img.height;
 694 |             if (!width || !height) {
 695 |                 return { dataUrl: rawDataUrl, originalSize: file.size, compressedSize: file.size, width: 0, height: 0 };
 696 |             }
 697 | 
 698 |             if (width > maxDimension || height > maxDimension) {
 699 |                 const scale = Math.min(maxDimension / width, maxDimension / height);
 700 |                 width = Math.max(1, Math.round(width * scale));
 701 |                 height = Math.max(1, Math.round(height * scale));
 702 |             }
 703 | 
 704 |             const canvas = document.createElement('canvas');
 705 |             canvas.width = width;
 706 |             canvas.height = height;
 707 |             const ctx = canvas.getContext('2d');
 708 |             if (!ctx) {
 709 |                 return { dataUrl: rawDataUrl, originalSize: file.size, compressedSize: file.size, width: width, height: height };
 710 |             }
 711 | 
 712 |             ctx.fillStyle = '#FFFFFF';
 713 |             ctx.fillRect(0, 0, width, height);
 714 |             ctx.drawImage(img, 0, 0, width, height);
 715 | 
 716 |             const dataUrl = canvas.toDataURL('image/jpeg', 0.84);
 717 |             const estBytes = Math.round((dataUrl.length - dataUrl.indexOf(',') - 1) * 0.75);
 718 | 
 719 |             return {
 720 |                 dataUrl: dataUrl,
 721 |                 originalSize: file.size,
 722 |                 compressedSize: estBytes,
 723 |                 width: width,
 724 |                 height: height
 725 |             };
 726 |         } catch (err) {
 727 |             return {
 728 |                 dataUrl: rawDataUrl,
 729 |                 originalSize: file.size,
 730 |                 compressedSize: file.size,
 731 |                 width: 0,
 732 |                 height: 0
 733 |             };
 734 |         }

 967 |         if (lat === null || lat === undefined || isNaN(lat) || String(lat).trim() === '') {
 968 |             return true;
 969 |         }
 970 |         const x = Number(lat);
 971 |         const y = Number(lng);
 972 |         const poly = SIES_CAMPUS_POLYGON;
 973 |         let inside = false;
 974 |         for (let i = 0, j = poly.length - 1; i < poly.length; j = i++) {
 975 |             const xi = poly[i][0], yi = poly[i][1];
 976 |             const xj = poly[j][0], yj = poly[j][1];
 977 |             const intersect = ((yi > y) !== (yj > y)) &&
 978 |                 (x < (xj - xi) * (y - yi) / (yj - yi) + xi);
